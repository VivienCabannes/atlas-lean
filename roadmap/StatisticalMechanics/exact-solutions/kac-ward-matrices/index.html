    noncomputable def ons_turnW (ω : ℂ) (μ ν : Fin 4) : ℂ :=
  if ν = μ then 1
  else if ν = μ + 1 then ω
  else if ν = μ + 2 then 0
  else ω⁻¹

    StatMech.Onsager.ons_spinDet

    The sector determinant: the product of ons_symbolDispersion β over the momenta (2π(i + a/2) / L, 2π(j + b/2) / L), i, j < L, for boundary phases a, b ∈ {0, 1}.

    noncomputable def ons_spinDet (L : ℕ) (β : ℝ) (a b : Fin 2) : ℝ :=
  ∏ i ∈ Finset.range L, ∏ j ∈ Finset.range L,
    ons_symbolDispersion β
      (2 * Real.pi * (i + (a.val : ℝ) / 2) / L)
      (2 * Real.pi * (j + (b.val : ℝ) / 2) / L)
    noncomputable def ons_symbolDispersion (β k₁ k₂ : ℝ) : ℝ :=
  (1 + Real.tanh β ^ 2) ^ 2 -
    2 * Real.tanh β * (1 - Real.tanh β ^ 2) * (Real.cos k₁ + Real.cos k₂)

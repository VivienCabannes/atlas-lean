    MetaMathlibExt.eisenstein_congr_sum_delta_pow_mul_eisenstein

    Let \(k \ge 12\) be even with \(k \ne 14\). For \(w, n \in \mathbb{N}\) let \(\sigma_{-1}(n)\) be \(1\) if \(n = 0\) and \(0\) otherwise; for \(w \ge 1\) let \(\sigma_{w-1}(0) = -B_w/(2w)\), where \(B_w\) is the \(w\)-th Bernoulli number, and \(\sigma_{w-1}(n) = \sum_{d \mid n} d^{w-1}\) for \(n \ge 1\). In \(\mathbb{Q}[[q]]\) let \(G_w = \sum_{n \ge 0} \sigma_{w-1}(n) q^n\), let \(E_w = G_w/\sigma_{w-1}(0)\) for even \(w\), and let \(\Delta = q \prod_{j \ge 1} (1 - q^j)^{24}\); for \(i, n \in \mathbb{N}\) write \(b_i(n)\) for the coefficient of \(q^n\) in \(\Delta^i E_{k-12i}\). Let \(N_w = 1\) for \(w \le 2\), let \(N_w\) be the absolute value of the numerator of \(\sigma_{w-1}(0)\) for \(w \ge 3\), and let \(N'_k\) be the largest divisor of \(N_k\) that is coprime to \(N_{k-12i}\) for every \(1 \le i \le \lfloor k/12 \rfloor\). For rationals \(x, y\) and a positive integer \(N\), \(x \equiv y \pmod{N}\) means that \((x - y)/N\) has denominator coprime to \(N\). Let \(m = \lfloor k/12 \rfloor - 1\) if \(k \equiv 2 \pmod{12}\) and \(m = \lfloor k/12 \rfloor\) otherwise, let \(C\) be the \(m \times m\) matrix \((b_i(j))_{1 \le i, j \le m}\), and let \((\gamma_1, \dots, \gamma_m) = (\sigma_{k-1}(1), \dots, \sigma_{k-1}(m))\, C^{-1}\). Then \(\sigma_{k-1}(n) \equiv \sum_{i=1}^m \gamma_i b_i(n) \pmod{N'_k}\) for every \(n \ge 0\), that is, \(G_k \equiv \sum_{i=1}^m \gamma_i \Delta^i E_{k-12i} \pmod{N'_k}\) coefficientwise.

    public theorem_wanted eisenstein_congr_sum_delta_pow_mul_eisenstein (k : ℕ) (hk : 12 ≤ k)
    (hke : Even k) (hk14 : k ≠ 14) :
    let σ₀ : ℕ → ℕ → ℚ := fun w n => if w = 0 then (if n = 0 then 1 else 0)
      else if n = 0 then -bernoulli w / (2 * w) else (ArithmeticFunction.sigma (w - 1) n : ℚ)
    let Δ : ℚ⟦X⟧ := X * ∏' j : ℕ, (1 - X ^ (j + 1)) ^ 24
    let b : ℕ → ℕ → ℚ := fun i n =>
      coeff n (Δ ^ i * PowerSeries.mk (σ₀ (k - 12 * i))) / σ₀ (k - 12 * i) 0
    let Nw : ℕ → ℕ := fun w => if w ≤ 2 then 1 else (σ₀ w 0).num.natAbs
    let N' : ℕ := ((Nat.divisors (Nw k)).filter fun d =>
      ∀ i ∈ Finset.Icc 1 (k / 12), Nat.Coprime d (Nw (k - 12 * i))).sup id
    let m : ℕ := if k % 12 = 2 then k / 12 - 1 else k / 12
    let C : Matrix (Fin m) (Fin m) ℚ := fun i j => b (i.val + 1) (j.val + 1)
    let γ : Fin m → ℚ := Matrix.vecMul (fun j => σ₀ k (j.val + 1)) C⁻¹
    ∀ n : ℕ, ((σ₀ k n - ∑ i : Fin m, γ i * b (i.val + 1) n) / N').den.Coprime N'

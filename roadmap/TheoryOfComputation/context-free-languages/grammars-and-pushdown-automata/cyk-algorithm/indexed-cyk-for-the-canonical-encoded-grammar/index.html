    def recognizes (termCost ntCost : ℕ) (start : N)
    (rs : List (ContextFreeRule T N)) (w : List T) : TimeM ℕ Bool :=
  match w with
  | [] =>
      let r := epsilon ntCost start rs
      ⟨r.ret, r.time + 1⟩
  | _ :: _ =>
      let a := w.toArray
      let tbl := table termCost ntCost a rs a.size
      let r := member ntCost start (lookup tbl.ret 0 a.size)
      ⟨r.ret, tbl.time + r.time + 3 * w.length + 12⟩

    def IsChomskyNormalForm (g : ContextFreeGrammar T) : Prop :=
  (∀ r ∈ g.rules,
    (∃ B C, r.output = [Symbol.nonterminal B, Symbol.nonterminal C]) ∨
    (∃ a, r.output = [Symbol.terminal a]) ∨
    (r.output = [] ∧ r.input = g.initial)) ∧
  ((∀ r ∈ g.rules, Symbol.nonterminal g.initial ∉ r.output) ∨
    ∀ r ∈ g.rules, r.output ≠ [])

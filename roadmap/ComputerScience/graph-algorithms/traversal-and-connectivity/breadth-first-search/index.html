    DefinitionList model of breadth-first search#✓fully proved

    Cslib.Algorithms.Lean.BreadthFirstSearch.bfsModel

    Given a finite directed graph \(G\) with ordered adjacency lists and a source vertex \(s\), the list model keeps a list of processed pairs and a queue of pairs \((v, d)\) of a vertex and a distance, starting with no processed pairs and the queue \([(s, 0)]\). It performs one round per vertex of \(G\): if the queue is empty, nothing changes; otherwise the front pair \((u, d)\) is removed from the queue and recorded as processed, and then each entry \(w\) of the adjacency list of \(u\), from left to right, is appended to the queue as \((w, d + 1)\) unless \(w\) already occurs in a processed or queued pair. The reported distance of a vertex \(v\) is the distance in the first pair for \(v\) in the list of processed pairs in processing order followed by the queue, or none if there is no such pair; the reported order lists the processed vertices in processing order.

    The model can be evaluated by kernel reduction, which is used to check concrete results.

    def bfsModel (g : DirectedAdjacency V) (s : V) : BFSResult V :=
  let initial : List (V × Nat) × List (V × Nat) := ([], [(s, 0)])
  let final := g.vertices.foldr (fun _ state =>
    match state.2 with
    | [] => state
    | entry :: queue =>
        (g.adjacency entry.1).foldl (fun state vertex =>
          if ((state.1.reverse ++ state.2).map Prod.fst).contains vertex then state
          else (state.1, state.2 ++ [(vertex, entry.2 + 1)]))
          (entry :: state.1, queue)) initial
  let entries := final.1.reverse ++ final.2
  { distance := fun v => (entries.find? fun entry => entry.1 == v).map Prod.snd
    order := final.1.reverse.map Prod.fst }

    Cslib.Algorithms.Lean.BreadthFirstSearch.bfs

    Given a finite directed graph \(G\) with ordered adjacency lists and a source vertex \(s\), breadth-first search keeps a FIFO queue of pairs \((v, d)\) of a vertex and a distance and a table of discovered vertices with their distances, starting with the queue \([(s, 0)]\) and \(s\) discovered at distance \(0\). It performs one round per vertex of \(G\): if the queue is empty, nothing changes; otherwise it dequeues the front pair \((u, d)\), records \(u\) as processed, and scans the adjacency list of \(u\) from left to right, enqueueing each entry \(w\) that is not yet discovered as \((w, d + 1)\) and discovering it at distance \(d + 1\). The result reports for each vertex its discovered distance, or none if it was never discovered, and lists the processed vertices in processing order.

    The table is a hash map, so the search runs in expected time linear in the number of vertices plus the total length of the adjacency lists.

    def bfs (g : DirectedAdjacency V) (s : V) : BFSResult V :=
  let initial : List (V × Nat) × (Std.Queue (V × Nat) × Std.HashMap V Nat) :=
    ([], ((Std.Queue.empty : Std.Queue (V × Nat)).enqueue (s, 0),
      (Std.HashMap.emptyWithCapacity 1).insert s 0))
  let final := Nat.fold g.vertices.length (fun _ _ state =>
    match state.2.1.dequeue? with
    | none => state
    | some (entry, queue) =>
        (g.adjacency entry.1).foldl (fun state vertex =>
          if state.2.2.contains vertex then state
          else (state.1,
            (state.2.1.enqueue (vertex, entry.2 + 1),
              state.2.2.insert vertex (entry.2 + 1))))
          (entry :: state.1, (queue, state.2.2))) initial
  { distance := fun v => final.2.2[v]?

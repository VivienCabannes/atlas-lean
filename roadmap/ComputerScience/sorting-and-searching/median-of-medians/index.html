    Cslib.Algorithms.Lean.TimeM.medianOfMediansGroups

    Given a list \(xs\) over a linear order, split \(xs\) into consecutive blocks of five elements, the last of which may be shorter. Sort each block \(g\) by insertion sort and return, in block order, the entry at position \(\lfloor (|g| - 1)/2 \rfloor\) (counting from \(0\)) of each sorted block, that is, its lower median. The cost is the total number of comparisons made by these insertion sorts.

    def medianOfMediansGroups {α : Type u} [LinearOrder α] (xs : List α) :
    TimeM Nat (List α) :=
  let rec loop : List (List α) → TimeM Nat (List α)
    | [] => pure []
    | group :: groups => do
        let sorted ← insertionSort group
        let tail ← loop groups
        match sorted[(group.length - 1) / 2]? with
        | none => pure tail
        | some x => pure (x :: tail)
  loop (xs.toChunks 5)
    def insertionSort : List α → TimeM Nat (List α)
  | [] => pure []
  | a :: xs => TimeM.bind (insertionSort xs) (insertionSortInsert a)

    DefinitionMedian-of-medians selection#✓fully proved

    Cslib.Algorithms.Lean.TimeM.medianOfMedians

    Given a list \(L\) over a linear order and a zero-based rank \(k\), median-of-medians selection returns either an element of \(L\) or nothing, charging one unit of cost per order comparison. If \(k \ge |L|\), where \(|L|\) is the length of \(L\), it returns nothing at no cost. Otherwise it runs the following recursive procedure on \((L, k)\) with a budget of \(|L|\) nested calls, returning nothing if the budget runs out. A list of length at most \(140\) is handled by deterministic quickselect at the current rank. A longer list is cut into consecutive groups of five entries, the last possibly shorter; each group is sorted by insertion sort and its lower median is kept, giving a list \(M\) of \(m\) medians. The procedure recursively selects the entry of \(M\) at rank \(\lfloor m/2 \rfloor\) as the pivot \(x\) (returning nothing if that call does), and splits the list by a three-way partition into the entries below \(x\), equal to \(x\) and above \(x\), with \(b\) entries below and \(e\) equal. If the current rank \(r\) satisfies \(r < b\) it recurses on the lower part with rank \(r\); if \(b \le r < b + e\) it returns \(x\); otherwise it recurses on the upper part with rank \(r - b - e\).

    def medianOfMedians {α : Type u} [LinearOrder α] (xs : List α) (k : Nat) :
    TimeM Nat (Option α) :=
  let rec loop : Nat → List α → Nat → TimeM Nat (Option α)
    | 0, _, _ => pure none
    | fuel + 1, ys, rank =>
        if ys.length ≤ 140 then quickselect ys rank else do
          let medians ← medianOfMediansGroups ys
          let pivot ← loop fuel medians (medians.length / 2)
          match pivot with
          | none => pure none
          | some x => do
              let parts ← partition3 x ys
              if rank < parts.below.length then
                loop fuel parts.below rank
              else if rank < parts.below.length + parts.equal.length then
                pure (some x)
              else
                loop fuel parts.above (rank - (parts.below.length + parts.equal.length))
  if k < xs.length then loop xs.length xs k else pure none

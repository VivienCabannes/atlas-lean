    Cslib.Algorithms.Lean.TimeM.quickselect

    Deterministic quickselect takes a list \(L\) over a linear order and a zero-based rank \(k\) and returns an element or no element, together with its cost, the number of element comparisons it makes. If \(k \ge |L|\), it returns no element at no cost. Otherwise it runs the following recursion, starting from the list \(L\) and the rank \(k\). On an empty list it returns no element. On a list with first entry \(p\) and remaining entries \(T\), and a rank \(r\), it forms the three-way partition of \(T\) around \(p\), with blocks \(B\), \(E\), \(A\) of entries below, equal to and above \(p\), at a cost of \(|T|\) comparisons. If \(r < |B|\), it recurses on \(B\) with rank \(r\); if \(|B| \le r < |B| + |E| + 1\), it returns \(p\); otherwise it recurses on \(A\) with rank \(r - (|B| + |E| + 1)\).

    The recursion carries a fuel counter that starts at \(|L|\) and drops by one per call, and it returns no element when the fuel is exhausted; since every recursive call is on a strictly shorter list, the fuel never runs out on a nonempty list.

    def quickselect (xs : List α) (k : Nat) : TimeM Nat (Option α) :=
  let rec loop : Nat → List α → Nat → TimeM Nat (Option α)
    | 0, _, _ => pure none
    | _ + 1, [], _ => pure none
    | fuel + 1, pivot :: tail, rank => do
        let parts ← partition3 pivot tail
        if rank < parts.below.length then
          loop fuel parts.below rank
        else if rank < parts.below.length + parts.equal.length + 1 then
          return some pivot
        else
          loop fuel parts.above
            (rank - (parts.below.length + parts.equal.length + 1))
  if k < xs.length then loop xs.length xs k else pure none
    def partition3 (pivot : α) : List α → TimeM Nat (ThreeWayPartition α)
  | [] => pure { below := [], equal := [], above := [] }
  | x :: xs => do
      ✓
      let parts ← partition3 pivot xs
      match cmp x pivot with
      | .lt => return { parts with below := x :: parts.below }
      | .eq => return { parts with equal := x :: parts.equal }
      | .gt => return { parts with above := x :: parts.above }

    Cslib.Algorithms.Lean.TimeM.quickselect_correct

    Let \(L\) be a list over a linear order and let \(k\) be a natural number. If quickselect on \(L\) with zero-based rank \(k\) returns an element \(x\), then \(x\) is an entry of \(L\), and, counting entries of \(L\) with multiplicity, the number of entries less than \(x\) is at most \(k\) and \(k\) is less than the number of entries at most \(x\).

    Thus \(x\) occupies zero-based position \(k\) in the sorted order of \(L\), also when \(x\) is repeated.

    theorem quickselect_correct {xs : List α} {k : Nat} {x : α}
    (hresult : (quickselect xs k).ret = some x) :
    x ∈ xs ∧
      xs.countP (fun y => decide (y < x)) ≤ k ∧
      k < xs.countP (fun y => decide (y ≤ x))

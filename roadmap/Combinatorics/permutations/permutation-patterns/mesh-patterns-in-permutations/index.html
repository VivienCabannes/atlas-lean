    MetaMathlibExt.meshPatternGeneratingFunction_eq_factorial_mul_prod_general

    Let \(R\) be a commutative semiring, let \(x \in R\), let \(0 < k \le n\) be natural numbers, and let \(P(m, r) \in R\) for natural numbers \(m, r\). Natural numbers are mapped into \(R\). Suppose that \(P(k-1, k) = (k-1)!\) and that \(P(m, k) = (x + (m - 1))\,P(m-1, k)\) for every natural number \(m \ge k\). Then

    \[P(n, k) = (k-1)! \prod_{i=0}^{n-k} \bigl(x + (k - 1 + i)\bigr).\]

    With \(R\) the polynomials over \(\mathbb{N}\) and \(x\) the variable, this is the product formula for the mesh-pattern counting polynomial.

    theorem meshPatternGeneratingFunction_eq_factorial_mul_prod_general {R : Type*} [CommSemiring R]
    (P : ℕ → ℕ → R) (n k : ℕ) (x : R)
    (hk : 0 < k) (hnk : k ≤ n)
    (hbase : P (k - 1) k = (Nat.factorial (k - 1) : R))
    (hstep : ∀ m : ℕ, k ≤ m →
      P m k = (x + ((m - 1 : ℕ) : R)) * P (m - 1) k) :
    P n k =
      (Nat.factorial (k - 1) : R) *
        ∏ i ∈ Finset.range (n - k + 1),
          (x + ((k - 1 + i : ℕ) : R))

    Let \(\sigma\) be a permutation of \(m\) elements, with positions numbered \(1, \dots, m\), let \(p\) be the position of the largest value of \(\sigma\), and let \(r\) be a natural number. A position \(i\) is a mesh-pattern match of \(\sigma\) with parameter \(r\) when \(\sigma(t) < \sigma(i)\) for every position \(t < i\) and at least \(r - 1\) positions \(t\) with \(i < t < p\) have \(\sigma(t) > \sigma(i)\).

    def MatchPred {m : ℕ} (r : ℕ) (σ : Equiv.Perm (Fin m)) (i : Fin m) : Prop :=
  ∃ maxIndex : Fin m,
    (∀ t : Fin m, σ t ≤ σ maxIndex) ∧
    r - 1 ≤ (Finset.univ.filter fun t : Fin m =>
      i < t ∧ t < maxIndex ∧ σ i < σ t).card ∧
    ∀ t : Fin m, t < i → σ t < σ i

    TheoremStirling-type recursion for the mesh-pattern counting polynomial#✓fully proved

    MetaMathlibExt.meshPatternCount_stirlingRecursion

    Let \(1 < k \le n\) and \(j > 0\). For a permutation \(\sigma\) of \(m\) elements, with positions numbered \(1, \dots, m\), let \(p\) be the position of its largest value and let \(N_r(\sigma)\) be the number of positions \(i\) such that \(\sigma(t) < \sigma(i)\) for all \(t < i\) and at least \(r - 1\) positions \(t\) with \(i < t < p\) have \(\sigma(t) > \sigma(i)\), plus \(1\) if \(p \ge r\). Let \(P_{m,r}(X) \in \mathbb{N}[X]\) be the sum of \(X^{N_r(\sigma)}\) over all permutations \(\sigma\) of \(m\) elements, and let \(c_{m,r}(s)\) be its coefficient of \(X^s\). Then \(c_{n,k}(j) = (n-1)\,c_{n-1,k}(j) + c_{n-1,k}(j-1)\), \(P_{n,k}(X) = (X + n - 1)\,P_{n-1,k}(X)\), \(c_{n,k}(0) = (n-1)\,c_{n-1,k}(0)\), and \(P_{k-1,k}(X) = (k-1)!\).

    theorem meshPatternCount_stirlingRecursion
    (k n j : ℕ) (hk : 1 < k) (hn : k ≤ n) (hj : 0 < j) :
    let isMeshMatch : {m : ℕ} → ℕ → Equiv.Perm (Fin m) → Fin m → Prop :=
      fun {m} r σ i =>
        ∃ maxIndex : Fin m,
          (∀ t : Fin m, σ t ≤ σ maxIndex) ∧
          r - 1 ≤ (Finset.univ.filter fun t : Fin m =>
            i < t ∧ t < maxIndex ∧ σ i < σ t).card ∧
          ∀ t : Fin m, t < i → σ t < σ i
    let zeroMatches : {m : ℕ} → ℕ → Equiv.Perm (Fin m) → Prop :=
      fun {m} r σ =>
        ∃ maxIndex : Fin m,
          (∀ t : Fin m, σ t ≤ σ maxIndex) ∧ r ≤ maxIndex.val + 1
    let meshCount : {m : ℕ} → ℕ → Equiv.Perm (Fin m) → ℕ :=
      fun {_m} r σ =>
        (Finset.univ.filter (isMeshMatch r σ)).card + if zeroMatches r σ then 1 else 0
    let P : ℕ → ℕ → Polynomial ℕ := fun m r =>
      ∑ σ : Equiv.Perm (Fin m), Polynomial.X ^ meshCount r σ
    let C : ℕ → ℕ → ℕ → ℕ := fun m r s => (P m r).coeff s
    C n k j = (n - 1) * C (n - 1) k j + C (n - 1) k (j - 1) ∧
      P n k = (Polynomial.X + Polynomial.C (n - 1)) * P (n - 1) k ∧
      C n k 0 = (n - 1) * C (n - 1) k 0 ∧
      P (k - 1) k = Polynomial.C (Nat.factorial (k - 1))

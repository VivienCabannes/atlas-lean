    Minuscule polynomials of type A

    DefinitionType-A minuscule polynomial#✓fully proved

    MetaMathlibExt.minusculePolynomialA

    For a natural number \(n\), the type-A minuscule polynomial \(N_n(x)\) is the polynomial with rational coefficients

    \[N_n(x) = \frac{1}{4n + 2} \sum_{k=1}^{n-1} k (n - k) \binom{2n + 2}{2k + 1} x^k.\]

    For \(n = 0\) the sum is empty, so \(N_0 = 0\).

    public noncomputable def minusculePolynomialA (n : ℕ) : ℚ[X] :=
  ∑ k ∈ Finset.Icc 1 (n - 1),
    C ((k : ℚ) * ((n : ℚ) - k) * ((2 * n + 2).choose (2 * k + 1) : ℚ) / (4 * n + 2)) * X ^ k

    TheoremBender's central limit theorem for polynomials with non-positive real zeros#●ready to prove

    MetaMathlibExt.bender_asymptotic_normality

    For each natural number \(n\) let \(r_{n,0}, \dots, r_{n,n-1}\) be non-negative real numbers, and let \(a(n,k)\) be the coefficient of \(x^k\) in the real polynomial \(A_n(x) = \prod_{i\lt n} (x + r_{n,i})\). Put \(\mu_n = \sum_{i\lt n} 1/(1 + r_{n,i})\) and \(\sigma_n^2 = \sum_{i\lt n} r_{n,i}/(1 + r_{n,i})^2\), and let \(\sigma_n \ge 0\) be the square root of \(\sigma_n^2\). Then for every \(n\), \(\sum_{k=0}^{n} k\, a(n,k) / \sum_{k=0}^{n} a(n,k) = \mu_n\) and \(\sum_{k=0}^{n} (k - \mu_n)^2 a(n,k) / \sum_{k=0}^{n} a(n,k) = \sigma_n^2\). Moreover, if \(\sigma_n^2 \to \infty\) as \(n \to \infty\), then for every real \(x\), \(\sum_{0 \le k \le n,\ k \le \mu_n + x \sigma_n} a(n,k) / \sum_{k=0}^{n} a(n,k)\) tends to \(\Phi(x)\) as \(n \to \infty\), where \(\Phi\) is the distribution function of the standard normal distribution.

    public theorem_wanted bender_asymptotic_normality (r : (n : ℕ) → Fin n → ℝ)
    (hr : ∀ n i, 0 ≤ r n i) :
    let a : ℕ → ℕ → ℝ := fun n k => (∏ i, (X + C (r n i))).coeff k
    let μ : ℕ → ℝ := fun n => ∑ i, 1 / (1 + r n i)
    let σ2 : ℕ → ℝ := fun n => ∑ i, r n i / (1 + r n i) ^ 2
    (∀ n, (∑ k ∈ Finset.range (n + 1), (k : ℝ) * a n k) /
          (∑ k ∈ Finset.range (n + 1), a n k) = μ n ∧
        (∑ k ∈ Finset.range (n + 1), ((k : ℝ) - μ n) ^ 2 * a n k) /
          (∑ k ∈ Finset.range (n + 1), a n k) = σ2 n) ∧
    (Tendsto σ2 atTop atTop → ∀ x : ℝ,
      Tendsto (fun n => (∑ k ∈ (Finset.range (n + 1)).filter (fun k : ℕ => (k : ℝ) ≤ μ n + x * √(σ2 n)),
            a n k) /
          (∑ k ∈ Finset.range (n + 1), a n k))
        atTop (𝓝 (ProbabilityTheory.cdf (ProbabilityTheory.gaussianReal 0 1) x)))
